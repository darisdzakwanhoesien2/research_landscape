\section{Phase 1: Constraint-Aware Extraction of ESG Claims}
\subsection{Objectives}
\begin{itemize}
  \item Build XBRL ingestion pipeline with taxonomy normalization.
  \item Fine-tune FinBERT for performative vs actionable ESG claims.
  \item Measure extraction reliability under structured noise.
\end{itemize}
\subsection{Research Gap}
\begin{itemize}
  \item No validated extraction pipelines exist for hierarchical financial filings.
\end{itemize}
\subsection{Problem Statement}
\begin{itemize}
  \item NLP systems cannot reliably validate extracted claims against financial structure.
\end{itemize}
\subsection{Research Questions}
\begin{itemize}
  \item What extraction accuracy is achievable on XBRL-derived text?
  \item How sensitive is extraction to structural normalization errors?
\end{itemize}
\subsection{Hypotheses}
\begin{itemize}
  \item Domain adaptation improves macro-F1 by ≥10\%.
  \item Structural noise degrades extraction nonlinearly.
\end{itemize}
\subsection{Expected Contributions}
\begin{itemize}
  \item Verified XBRL extraction pipeline.
  \item Constraint violation detection benchmark.
\end{itemize}

  \item Quantify violation reduction.
\end{itemize}
\subsection{Research Gap}

  \item Neural models lack formal constraint validation mechanisms.
\end{itemize}
\subsection{Problem Statement}

  \item Statistical predictions violate regulatory consistency rules.
\end{itemize}
\subsection{Research Questions}

  \item Can constraints reduce violations without accuracy loss?
\end{itemize}
\subsection{Hypotheses}

  \item Constraint enforcement reduces violations by ≥30\%.
\end{itemize}
\subsection{Expected Contributions}

  \item Formal regulatory constraint ontology.
\end{itemize}
